-- ============================================================
-- SCAN LOGIC
-- ============================================================
local scanLogLines = {}  -- holds strings

local function logLine(str)
    table.insert(scanLogLines, str)
end

local function rebuildLogText()
    LogBox.Text = table.concat(scanLogLines, "\n")
end

local function getFullPath(obj)
    -- Returns a path usable with game:GetService() or workspace lookups
    local parts = {}
    local current = obj
    while current and current ~= game do
        table.insert(parts, 1, current.Name)
        current = current.Parent
    end
    -- Determine root service
    local rootName = parts[1]
    local rest = {}
    for i = 2, #parts do
        rest[#rest+1] = parts[i]
    end
    return "game:GetService(\"" .. rootName .. "\")." .. table.concat(rest, ".")
end

local function scanGame()
    scanLogLines = {}
    logLine("=== SCAN RESULTS ===")
    logLine("Time: " .. os.date("%Y-%m-%d %H:%M:%S"))
    logLine("")

    local clickDetectors = {}
    local touchInterests = {}

    -- We scan all descendants of game (workspace, ReplicatedStorage accessible etc.)
    -- Use pcall in case we hit restricted instances.
    local ok, err = pcall(function()
        for _, obj in ipairs(game:GetDescendants()) do
            if obj:IsA("ClickDetector") then
                table.insert(clickDetectors, obj)
            elseif obj:IsA("TouchInterest") then
                table.insert(touchInterests, obj)
            end
        end
    end)

    if not ok then
        logLine("[!] Partial scan (restricted areas skipped): " .. tostring(err))
    end

    logLine(string.format("Found %d ClickDetector(s)", #clickDetectors))
    logLine("---------------------------------")
    if #clickDetectors == 0 then
        logLine("  (none)")
    else
        for i, cd in ipairs(clickDetectors) do
            local path = getFullPath(cd)
            logLine(string.format("[%d] TYPE: ClickDetector", i))
            logLine("    PATH: " .. path)
            -- Include the parent that actually holds the detector
            if cd.Parent then
                logLine("    PARENT: " .. getFullPath(cd.Parent))
            end
            logLine("")
        end
    end

    logLine(string.format("Found %d TouchInterest(s)", #touchInterests))
    logLine("---------------------------------")
    if #touchInterests == 0 then
        logLine("  (none)")
    else
        for i, ti in ipairs(touchInterests) do
            local path = getFullPath(ti)
            logLine(string.format("[%d] TYPE: TouchInterest", i))
            logLine("    PATH: " .. path)
            if ti.Parent then
                logLine("    PARENT: " .. getFullPath(ti.Parent))
            end
            logLine("")
        end
    end

    logLine("=== END OF SCAN ===")

    rebuildLogText()

    return #clickDetectors, #touchInterests
end

ScanBtn.MouseButton1Click:Connect(function()
    StatusLabel.Text = "Scanning..."
    StatusLabel.TextColor3 = CONFIG.AccentColor
    -- small delay so UI updates before heavy scan
    task.wait(0.05)
    local cd, ti = scanGame()
    StatusLabel.Text = string.format("Done. %d ClickDetector(s), %d TouchInterest(s).", cd, ti)
    StatusLabel.TextColor3 = CONFIG.CopyColor
end)

local function setClipboard(text)
    -- Modern clipboard API (setclipboard is only in exploits; this is the
    -- legit UserInputService method for plugins/Studio - in-game it's stubbed
    -- to prevent abuse. We still call it; on mobile it silently fails.)
    local success, err = pcall(function()
        if setclipboard then
            setclipboard(text)
        end
    end)
    -- Roblox does not expose a public client clipboard API for regular games.
    -- As a fallback we select the text so the user can copy manually.
    LogBox:CaptureFocus()
    LogBox.CursorPosition = -1
    LogBox.SelectionStart = 1
    LogBox.SelectionStart = #LogBox.Text + 1
    return success
end

CopyBtn.MouseButton1Click:Connect(function()
    if #scanLogLines == 0 then
        StatusLabel.Text = "Nothing to copy - run a scan first."
        StatusLabel.TextColor3 = CONFIG.CloseColor
        return
    end

    local text = table.concat(scanLogLines, "\n")

    local copied = false
    -- Try exploit/Studio clipboard if available
    if typeof(setclipboard) == "function" then
        local ok = pcall(function() setclipboard(text) end)
        copied = ok
    end

    -- Fallback: highlight text inside the log for manual copy
    LogBox:CaptureFocus()
    task.wait(0.05)
    LogBox.CursorPosition = 1
    LogBox.SelectionStart = 1

    if copied then
        StatusLabel.Text = "Log copied to clipboard!"
        StatusLabel.TextColor3 = CONFIG.CopyColor
    else
        StatusLabel.Text = "Clipboard unavailable - text selected (long-press to copy)."
        StatusLabel.TextColor3 = CONFIG.AccentColor
    end
end)

CloseBtn.MouseButton1Click:Connect(function()
    ScreenGui:Destroy()
end)
